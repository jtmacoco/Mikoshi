# Example 1

**Input:** 
```python
["MyHashMap", "put", "put", "get", "get", "put", "get", "remove", "get"]
[[], [1, 1], [2, 2], [1], [3], [2, 1], [2], [2], [2]]
```

**Output:** 
`[null, null, null, 1, -1, null, 1, null, -1]`

**Explanation:**

```
MyHashMap myHashMap = new MyHashMap();
myHashMap.put(1, 1); // The map is now [[1,1]]
myHashMap.put(2, 2); // The map is now [[1,1], [2,2]]
myHashMap.get(1);    // return 1, The map is now [[1,1], [2,2]]
myHashMap.get(3);    // return -1 (i.e., not found), The map is now [[1,1], [2,2]]
myHashMap.put(2, 1); // The map is now [[1,1], [2,1]] (i.e., update the existing value)
myHashMap.get(2);    // return 1, The map is now [[1,1], [2,1]]
myHashMap.remove(2); // remove the mapping for 2, The map is now [[1,1]]
myHashMap.get(2);    // return -1 (i.e., not found), The map is now [[1,1]]

```

# Solution

>[!important] Intuition
> Brute force use loops. Efficient way buckets with a fixed size. 
> Bucket is more efficient since use a fixed size with a simple hash
> Hashing allows to have multiple keys in the same bucket 

```python title=brute-force
class MyHashMap:

    def __init__(self):
        self.hashMap = []

    def put(self, key: int, value: int) -> None:
        if not self.hashMap:
            self.hashMap.append([key,value])
        else:
            for i in range(len(self.hashMap)):
                hm_key = self.hashMap[i][0]
                if hm_key == key:
                    self.hashMap[i][1] = value
                    return
            self.hashMap.append([key,value])
            
        

    def get(self, key: int) -> int:
        for i in range(len(self.hashMap)):
            hm_key = self.hashMap[i][0]
            if hm_key == key:
                return self.hashMap[i][1]
        return -1

    def remove(self, key: int) -> None:
        for i in range(len(self.hashMap)):
            hm_key = self.hashMap[i][0]
            if hm_key == key:
                del self.hashMap[i]
                break
        


# Your MyHashMap object will be instantiated and called as such:
# obj = MyHashMap()
# obj.put(key,value)
# param_2 = obj.get(key)
# obj.remove(key)
```

```python title=bucket
class MyHashMap:

    def __init__(self):
        self.size = 1000
        self.buckets = [[] for _ in range(self.size)]

    def _hash(self, key:int) -> int:
        return key%self.size
    def put(self, key: int, value: int) -> None:
        idx = self._hash(key)
        bucket = self.buckets[idx]
        for i, (k,v) in enumerate(bucket):
            if key == k:
                bucket[i] = (key,value)
                return
        bucket.append((key,value))


    def get(self, key: int) -> int:
        idx = self._hash(key)
        bucket = self.buckets[idx]
        for k,v in bucket:
            if key == k:
                return v
        else:
            return -1
        

    def remove(self, key: int) -> None:
        idx = self._hash(key)
        bucket = self.buckets[idx]
        for i, (k,v) in enumerate(bucket):
            if key == k:
                bucket.pop(i)
                return
        


# Your MyHashMap object will be instantiated and called as such:
# obj = MyHashMap()
# obj.put(key,value)
# param_2 = obj.get(key)
# obj.remove(key)
```
